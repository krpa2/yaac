# Every ../assets/*.mid is embedded in the firmware and listed in a generated
# playlist (midi_playlist.c). Re-run the build after adding/removing files.
set(MIDI_FILES "")
set(PLAYLIST_C "")
# ESP-IDF evaluates this file once in script mode to collect requirements; only
# glob and generate in the real configure pass.
if(NOT CMAKE_BUILD_EARLY_EXPANSION)
    file(GLOB MIDI_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_LIST_DIR}/../assets/*.mid")
    list(SORT MIDI_FILES)

    set(PLAYLIST_C "${CMAKE_CURRENT_BINARY_DIR}/midi_playlist.c")
    set(decls "")
    set(entries "")
    foreach(f ${MIDI_FILES})
        get_filename_component(name "${f}" NAME)
        string(MAKE_C_IDENTIFIER "${name}" id)     # same symbol naming as EMBED_FILES
        string(APPEND decls "extern const uint8_t ${id}_start[] asm(\"_binary_${id}_start\");\n")
        string(APPEND decls "extern const uint8_t ${id}_end[] asm(\"_binary_${id}_end\");\n")
        string(APPEND entries "    {\"${name}\", ${id}_start, ${id}_end},\n")
    endforeach()
    list(LENGTH MIDI_FILES n_midi)
    file(CONFIGURE OUTPUT "${PLAYLIST_C}" CONTENT
    "/* Generated by main/CMakeLists.txt from the .mid files in ../assets - do not edit. */
    #include \"midi_playlist.h\"
    ${decls}
    const midi_playlist_entry_t midi_playlist[] = {
    ${entries}    {0, 0, 0},
    };
    const size_t midi_playlist_count = ${n_midi};
    ")
endif()

# The Imperial March score is shared with the ../sound project.
idf_component_register(SRCS "main.c" "amy_player.c" "song_korobeiniki.c" "song_imperial.c"
                            "midi_file.c" "midi_player.c" "${PLAYLIST_C}"
                            "../../sound/main/imperial_march.c"
                       INCLUDE_DIRS "." "../../sound/main"
                       EMBED_FILES ${MIDI_FILES}
                       REQUIRES amy board_power driver esp_timer)
